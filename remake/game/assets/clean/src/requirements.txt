#   ~/work/simant-venv/bin/python make_sprites.py
# Recreate it anywhere with:
#   python3 -m venv ~/work/simant-venv
#   ~/work/simant-venv/bin/python -m pip install -r remake/game/assets/clean/src/requirements.txt
#
# Unpinned on purpose, matching .github/workflows/build.yml: the .pix files are plain text and the
